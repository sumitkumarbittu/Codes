SELECT c.Customer_name,
       p.Product_name,
       o.Order_Date,
       o.Amount
FROM CUSTOMERS c
JOIN ORDERS o ON c.Customer_Id = o.Customer_Id
JOIN PRODUCT p ON p.Product_Id = o.Product_Id
WHERE p.Product_price > 500;

SELECT c.Customer_name
FROM CUSTOMERS c
LEFT JOIN ORDERS o ON c.Customer_Id = o.Customer_Id
WHERE o.Order_Id IS NULL;

CREATE VIEW Order_details_100 AS
SELECT c.Customer_Id,
       c.Customer_name,
       c.City,
       o.Product_Id,
       o.Order_Date,
       o.Amount
FROM CUSTOMERS c
JOIN ORDERS o ON c.Customer_Id = o.Customer_Id
WHERE o.Order_Date BETWEEN DATE '2025-06-01' AND DATE '2025-06-30'
WITH READ ONLY;

SELECT * FROM Order_details_100;

CREATE OR REPLACE TRIGGER check_product_price
BEFORE INSERT ON PRODUCT
FOR EACH ROW
BEGIN
    IF :NEW.Product_price < 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Invalid price! Price cannot be negative.');
    END IF;
END;
