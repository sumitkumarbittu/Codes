-- Sequence for EmployeeID
CREATE SEQUENCE emp_seq START WITH 1 INCREMENT BY 1;

-- EMPLOYEES TABLE
CREATE TABLE EMPLOYEES(
    EmployeeID NUMBER PRIMARY KEY,
    Name VARCHAR2(30),
    DepartmentID NUMBER,
    Age NUMBER,
    Salary NUMBER,
    ProjectID NUMBER
);

-- DEPARTMENTS TABLE
CREATE TABLE DEPARTMENTS(
    DepartmentID NUMBER PRIMARY KEY,
    DepartmentName VARCHAR2(30)
);

-- PROJECTS TABLE
CREATE TABLE PROJECTS(
    ProjectID NUMBER PRIMARY KEY,
    ProjectName VARCHAR2(50),
    No_of_Hours NUMBER
);

CREATE OR REPLACE TRIGGER check_age
BEFORE INSERT ON EMPLOYEES
FOR EACH ROW
BEGIN
    IF :NEW.Age < 18 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Employee age cannot be less than 18');
    END IF;
END;
